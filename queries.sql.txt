 1. Sample Data Check
SELECT * FROM superstore LIMIT 10;

 2. Filtering Queries
SELECT * FROM superstore WHERE field2 LIKE '%West%';
SELECT * FROM superstore WHERE field2 LIKE '%Furniture%';
SELECT * FROM superstore WHERE field2 LIKE '%Technology%';

 3. Aggregations
SELECT AVG(field1) AS AvgValue FROM superstore;
SELECT SUM(field1) AS TotalValue FROM superstore;

4. Top Results
SELECT field2, COUNT(*) AS Occurrences
FROM superstore
GROUP BY field2
ORDER BY Occurrences DESC
LIMIT 5;

 5. Monthly Trend
SELECT substr(field2,1,7) AS Month, SUM(field1) AS MonthlyValue
FROM superstore
GROUP BY Month
ORDER BY Month;

 6. Duplicate Check
SELECT field2, COUNT(*) 
FROM superstore
GROUP BY field2
HAVING COUNT(*) > 1;

 7. Validation
SELECT COUNT(*) FROM superstore;
SELECT COUNT(*) FROM superstore WHERE field1 IS NULL;
